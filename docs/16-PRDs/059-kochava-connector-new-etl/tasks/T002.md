using System.Text;
using Aim.DataIngestion.Shared.Utils;

namespace Aim.DataIngestion.SqlBuilders.DataConsolidation.Redshift;

public class FullOuterJoinQueryBuilder : BaseJoinQueryBuilder
{
    public static string GetConsolidatedTableQuery(string primaryTable, string priorConsolidationTable, string targetConsolidationTable, List<string> regions, List<string>? networksToJoinOnCampaignId = null)
    {
        var sql = BuildBaseQuery(targetConsolidationTable);
        sql.Append(GetSelectClause("p", "s", regions));
        sql.Append(GetFromClause(priorConsolidationTable, "p", primaryTable, "s", regions, networksToJoinOnCampaignId));
        FinalizeQuery(sql);
        return sql.ToString();
    }

    private static string GetSelectClause(string primaryTableAlias, string secondaryTableAlias, List<string> regions)
    {
        var sql = new StringBuilder();
        sql.AppendLine("SELECT");
        AppendIngestionGroupingKey(sql);
        AppendDimensions(sql, primaryTableAlias, secondaryTableAlias);
        AppendMetrics(sql, primaryTableAlias, secondaryTableAlias);
        AppendMetadata(sql);
        RedshiftSqlHelper.RemoveTrailingComma(sql);
