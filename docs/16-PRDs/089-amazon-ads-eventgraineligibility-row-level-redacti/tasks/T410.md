From `cmd/bqgoldenutil/subcmd/account_golden_images.go`:
```
package subcmd

import (
	"cloud.google.com/go/bigquery"
	"github.com/Kochava/bqgolden"
	"github.com/Kochava/bqgolden/cmd/bqgoldenutil/config"
	"github.com/Kochava/bqgolden/pkg/rowaccesspolicy"
)

func AccountGoldenImages(config *config.AccountGoldenImage, kochavaQueryBigQueryClient *bigquery.Client, apiClient *rowaccesspolicy.Client) []bqgolden.AccountGoldenImage {
	var accountGoldenImages []bqgolden.AccountGoldenImage

	if config.CoreAccount {
		accountGoldenImages = append(accountGoldenImages, bqgolden.CoreAccountGoldenImage(kochavaQueryBigQueryClient, apiClient))
	}
	if config.CoreAccountRestricted {
		accountGoldenImages = append(accountGoldenImages, bqgolden.CoreAccountRestrictedGoldenImage(kochavaQueryBigQueryClient, apiClient))
	}
	if config.CoreAccountUnredacted {
		accountGoldenImages = append(accountGoldenImages, bqgolden.CoreAccountUnredactedGoldenImage(kochavaQueryBigQueryClient, apiClient))
	}

	// AllAccountTables defaults to true so that running with no dataset flags at all still
	// syncs everything, but it must not override an explicit single/partial dataset selection
	// above - otherwise passing just one dataset flag is silently ignored in favor of "all".
```
