```
using System.Collections.Generic;
using System.Threading.Tasks;
using Kochava.Aim.Portal.DataAccess.Mongo.Models;

namespace Kochava.Aim.Portal.DataAccess.Mongo.DataLayers.Interfaces;

public interface IAimMetricsDataLayer : IMongoDataLayer {

	/// <summary>Gets the non-deleted AIM Metric options, ordered by Label.</summary>
	Task<IList<AimMetric>> GetAsync();

	/// <summary>
	/// Idempotently inserts the engineering-authored starting seed values using SetOnInsert, so a
	/// value Data Engineering has already edited (Label) or soft-deleted (Deleted) is never
	/// clobbered by a redeploy -- only a genuinely missing Value gets inserted. Call once at
	/// startup (mirrors IOnboardingSessionDataLayer.EnsureIndexesAsync's wiring in Startup.cs).
	/// </summary>
	Task EnsureSeedDataAsync();

	/// <summary>
	/// Creates a unique index on Value, so a concurrent seed-upsert race across two pods starting
	/// simultaneously can't leave two documents with the same Value. Call once at startup.
	/// </summary>
	Task EnsureIndexesAsync();
```
