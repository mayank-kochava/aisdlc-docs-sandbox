```
public class BsonDecimalSerializer : DecimalSerializer, IBsonSerializationProvider {

	public IBsonSerializer GetSerializer(Type type) => type == typeof(decimal) ? this : null;

	public override void Serialize(BsonSerializationContext context, BsonSerializationArgs args, decimal value) { context.Writer.WriteDouble((double)value); }

	public override decimal Deserialize(BsonDeserializationContext context, BsonDeserializationArgs args) {
		try {
			return Convert.ToDecimal(context.Reader.ReadDouble());
		} catch (OverflowException) {
			return decimal.MaxValue;
		}
	}

}
```

```
using System.Collections.Generic;
using MongoDB.Bson;
using MongoDB.Bson.Serialization;
using MongoDB.Bson.Serialization.Serializers;

namespace Kochava.Aim.Portal.DataAccess.Mongo;

/// <summary>
/// Handles MongoDB fields that can be either a string array or an empty object {}.
/// Returns an empty list when the BSON value is not an array.
/// </summary>
public class BsonStringListSerializer : SerializerBase<List<string>> {

	public override List<string> Deserialize(BsonDeserializationContext context, BsonDeserializationArgs args) {
		var bsonType = context.Reader.GetCurrentBsonType();

		if (bsonType == BsonType.Array) {
			var result = new List<string>();
			context.Reader.ReadStartArray();
			while (context.Reader.ReadBsonType() != BsonType.EndOfDocument) {
				result.Add(context.Reader.ReadString());
			}
			context.Reader.ReadEndArray();
			return result;
		}
```
