```
﻿using System;
using MongoDB.Bson.Serialization;
using MongoDB.Bson.Serialization.Serializers;

namespace Kochava.Aim.Portal.DataAccess.Mongo;

public class BsonDecimalSerializer : DecimalSerializer, IBsonSerializationProvider {

	public IBsonSerializer GetSerializer(Type type) => type == typeof(decimal) ? this : null;

	public override void Serialize(BsonSerializationContext context, BsonSerializationArgs args, decimal value) { context.Writer.WriteDouble((double)value); }

	public override decimal Deserialize(BsonDeserializationContext context, BsonDeserializationArgs args) {
		try {
			return Convert.ToDecimal(context.Reader.ReadDouble());
		} catch (OverflowException) {
			return decimal.MaxValue;
		}
	}

}
```

```
using MongoDB.Bson;
using MongoDB.Bson.Serialization;
using MongoDB.Bson.Serialization.Serializers;

namespace Kochava.Aim.Portal.DataAccess.Mongo;

/// <summary>
/// Reads a nullable bool from whatever BSON type the aim-incremental-service Python pipeline
/// actually wrote instead of throwing — a non-boolean/non-string type (or an unparsable string)
/// deserializes to null rather than failing the whole document.
/// </summary>
public class BsonFlexNullableBoolSerializer : SerializerBase<bool?> {

	public override bool? Deserialize(BsonDeserializationContext context, BsonDeserializationArgs args) {
		var reader = context.Reader;
		switch (reader.CurrentBsonType) {
			case BsonType.Boolean: return reader.ReadBoolean();
			case BsonType.Int32: return reader.ReadInt32() != 0;
			case BsonType.String: return bool.TryParse(reader.ReadString(), out var parsed) ? parsed : null;
			case BsonType.Null: reader.ReadNull(); return null;
			case BsonType.Undefined: reader.ReadUndefined(); return null;
			default: reader.SkipValue(); return null;
		}
	}
```
